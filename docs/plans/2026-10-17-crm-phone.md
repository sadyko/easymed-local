# Телефон внутри CRM (CRM_PHONE_V1) — план реализации

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Рабочее место телефониста (бывший EasyPhone на порту 8020) становится выдвижной панелью
раздела «CRM · Заявки» по образцу владельца; отдельная программа удаляется.

**Architecture:** Четыре маршрута `phone/index.js` переезжают в RPC-обработчики нового файла
`server/services/rpc/phone.js` за теми же тремя воротами (`crm.dial`, `crm.calls`, `crm.recording`).
Клиент — новый модуль `views/crm-phone.js` (панель на `document.body`, состояние в памяти модуля,
опрос раз в 7 с, снятие при уходе из CRM) плюс чистая логика `crm-phone-logic.js`. CRM-доска
получает одну кнопку в шапке. Набор и запись идут через уже существующие `telephony_dial` и
`telephony_call_recording`.

**Tech Stack:** Node 20+ (`node:test`), Express, better-sqlite3; браузер — ES-модули без сборки,
`h()` из `ui.js`, `tr()/trf()` из `i18n.js`, стенд `__tests__/crm-harness.mjs`.

Спецификация: `docs/specs/2026-10-17-crm-phone-design.md`.

## Правила работы

- В клоне параллельно работает другая сессия (META_LEADS_V1, MOIZVONKI_V2, SETTINGS_HUB_LIST_V1).
  Общие файлы (`admin.js`, `rpc/index.js`, `i18n-strings.js`, `admin-views.css`, `server/index.js`,
  `RELEASE_NOTES.md`) правятся ТОЛЬКО точечными вставками. `git add` — только свои пути.
- `npm test` целиком — не раньше, чем соседняя сессия сообщит, что её прогон окончен; до того —
  только точечные `node --test <файл>`.
- После сборки — СТОП: владелец смотрит dev (:8000). Ни push, ни тега.

```js
// CRM_PHONE_V1 (2026-10-17) — ТЕЛЕФОН ВНУТРИ CRM: четыре RPC рабочего места телефониста.
//
// Перенос phone/index.test.js (CALLCENTER_OPERATOR_V1): пять рабочих мест, каждое со
// СВОИМ набором прав, и один звонок с записью. Каждый вызов обязан спрашивать СВОЁ
// право, а ссылка на запись — приходить только тому, кому прослушивание выдано.
// Отдельной программы больше нет, но дыра, которую эти тесты закрывали, никуда не
// делась: журнал звонков — вся история обращений клиники, и право ПОЗВОНИТЬ его
// не открывает.
import { test } from 'node:test';
import assert from 'node:assert/strict';
import { openDb } from '../../db/connection.js';
import { migrate } from '../../db/migrate.js';
import { telephonyPhoneMe, telephonyMyExtension, telephonyExtensions, telephonyCallLog } from './phone.js';
import { RpcError } from './telephony.js';
import { getRpc } from './index.js';

function setGrants(db, role, grants) {
  const row = db.prepare('SELECT permissions FROM role_permissions WHERE role = ?').get(role);
  const perms = row && row.permissions ? JSON.parse(row.permissions) : { sections: [], levels: {} };
  perms.grants = { ...(perms.grants || {}), ...grants };
  if (row) db.prepare('UPDATE role_permissions SET permissions = ? WHERE role = ?').run(JSON.stringify(perms), role);
  else db.prepare('INSERT INTO role_permissions (role, permissions) VALUES (?, ?)').run(role, JSON.stringify(perms));
}

// Пять рабочих мест, каждое со СВОИМ набором прав, — чтобы «пустил» нельзя было
// спутать с «не пустил» по соседнему праву.
function seed() {
  const db = openDb(':memory:');
  migrate(db);
  const mk = db.prepare('INSERT INTO users (id, username, password_hash, full_name, role) VALUES (?,?,?,?,?)');
  for (const [id, name, full, role] of [
    [1, 'adm', 'Администратор', 'admin'],
    [2, 'dialer', 'Только набор', 'nurse'],
    [3, 'watcher', 'Только журнал', 'lab'],
    [4, 'listener', 'Журнал и записи', 'cashier'],
    [5, 'nobody', 'Без телефонии', 'doctor'],
  ]) mk.run(id, name, 'x', full, role);
  // Миграция 141 уже записала этим ролям явные «Нет» — выдаём поверх ровно то,
  // что проверяем.
  setGrants(db, 'nurse',   { 'crm.dial': 'edit', 'crm.calls': 'none', 'crm.recording': 'none' });
  setGrants(db, 'lab',     { 'crm.dial': 'none', 'crm.calls': 'view', 'crm.recording': 'none' });
  setGrants(db, 'cashier', { 'crm.dial': 'none', 'crm.calls': 'view', 'crm.recording': 'edit' });
  db.prepare(`INSERT INTO calls (general_call_id, started_at, call_type, external_number, internal_number, billsec, recording_url)
              VALUES ('c-1', '2026-09-20T08:00:00Z', 0, '998901112233', '101', 42, 'https://rec/1.mp3')`).run();
  return db;
}
const U = {
  adm: { id: 1, role: 'admin' }, dialer: { id: 2, role: 'nurse' }, watcher: { id: 3, role: 'lab' },
  listener: { id: 4, role: 'cashier' }, nobody: { id: 5, role: 'doctor' },
};
const denied = (fn, status = 403) => assert.throws(fn, (e) => e instanceof RpcError && e.status === status);

test('все четыре вызова зарегистрированы в реестре RPC', () => {
  for (const n of ['telephony_phone_me', 'telephony_my_extension', 'telephony_extensions', 'telephony_call_log']) {
    assert.equal(typeof getRpc(n), 'function', n);
  }
});

test('без прав — отказ на всём', async () => {
  const db = seed();
  denied(() => telephonyPhoneMe(db, {}, U.nobody));
  denied(() => telephonyCallLog(db, {}, U.nobody));
  denied(() => telephonyMyExtension(db, { extension: '101' }, U.nobody));
  await assert.rejects(() => telephonyExtensions(db, {}, U.nobody), (e) => e instanceof RpcError && e.status === 403);
});

// ГЛАВНОЕ. Журнал звонков — это вся история обращений клиники: номера, время,
// кто звонил и ссылка на запись разговора. Право ПОЗВОНИТЬ его не открывает.
test('журнал звонков открывает право «Журнал звонков», а не право «Позвонить»', () => {
  const db = seed();
  assert.throws(() => telephonyCallLog(db, {}, U.dialer), (e) => e.status === 403 && /«Журнал звонков»/.test(e.message),
    'человеку с одним лишь набором номера отдали весь журнал звонков клиники');
  assert.equal(telephonyCallLog(db, {}, U.watcher).calls.length, 1);
  assert.equal(telephonyCallLog(db, {}, U.adm).calls.length, 1);
});

test('набор и своя трубка — право «Позвонить», а не право журнала', async () => {
  const db = seed();
  denied(() => telephonyMyExtension(db, { extension: '101' }, U.watcher));
  await assert.rejects(() => telephonyExtensions(db, {}, U.watcher), (e) => e.status === 403);
  assert.deepEqual(telephonyMyExtension(db, { extension: '101' }, U.dialer), { ok: true, extension: '101' });
  assert.deepEqual(await telephonyExtensions(db, {}, U.dialer), { extensions: [], error: '' });
});

// Экран рисует «Позвонить», журнал и «Прослушать» по трём флагам — тем же воротам,
// что у самих вызовов: кнопка, которую сервер всё равно откажет, не рисуется.
test('рабочее место (me): три флага прав честные', () => {
  const db = seed();
  const flags = (u) => { const m = telephonyPhoneMe(db, {}, u); return [m.may_dial, m.may_log, m.may_hear]; };
  assert.deepEqual(flags(U.dialer),   [true,  false, false]);
  assert.deepEqual(flags(U.watcher),  [false, true,  false]);
  assert.deepEqual(flags(U.listener), [false, true,  true]);
  assert.deepEqual(flags(U.adm),      [true,  true,  true]);
  const me = telephonyPhoneMe(db, {}, U.dialer);
  assert.equal(me.full_name, 'Только набор');
  assert.equal(me.extension, '');
  assert.equal(me.line, null, 'линии в тесте нет');
});

// ССЫЛКА НА ЗАПИСЬ — ОТДЕЛЬНОЕ ПРАВО, И В ЖУРНАЛЕ ЕЁ БЫТЬ НЕ ДОЛЖНО. Адрес записи
// у Binotel и «Моих Звонков» прямой: уехав в журнал, он даёт голос пациента
// любому, кто журнал открыл. Флаг has_recording остаётся — кнопка рисуется по
// нему и получает честный отказ.
test('запись разговора приходит только тому, кому выдано прослушивание', () => {
  const db = seed();
  const seen = telephonyCallLog(db, {}, U.watcher);
  assert.equal(seen.may_hear, false);
  assert.equal(seen.calls[0].recording_url, null, 'ссылка на запись уехала в журнал мимо права «Прослушать»');
  assert.equal(seen.calls[0].has_recording, true, 'журнал скрыл сам факт записи — кнопке не из чего взяться');
  const heard = telephonyCallLog(db, {}, U.listener);
  assert.equal(heard.may_hear, true);
  assert.equal(heard.calls[0].recording_url, 'https://rec/1.mp3', 'выданное право прослушивания не отдало запись');
  assert.equal(telephonyCallLog(db, {}, U.adm).calls[0].recording_url, 'https://rec/1.mp3');
});

test('своя трубка: цифры не длиннее двенадцати, пишется ТОЛЬКО себе', () => {
  const db = seed();
  denied(() => telephonyMyExtension(db, { extension: 'abc' }, U.dialer), 400);
  denied(() => telephonyMyExtension(db, { extension: '1234567890123' }, U.dialer), 400);
  telephonyMyExtension(db, { extension: '102' }, U.dialer);
  const ext = (id) => db.prepare('SELECT pbx_extension FROM users WHERE id = ?').get(id).pbx_extension;
  assert.equal(ext(2), '102');
  assert.equal(ext(1), null, 'трубка назначилась чужой строке');
  assert.equal(telephonyPhoneMe(db, {}, U.dialer).extension, '102');
  telephonyMyExtension(db, { extension: '' }, U.dialer);
  assert.equal(ext(2), null, 'пустой выбор должен снимать трубку, а не писать пустую строку');
});

test('журнал: предел зажат в 1..200, по умолчанию 60; поля для экрана на месте', () => {
  const db = seed();
  const add = db.prepare(`INSERT INTO calls (general_call_id, started_at, call_type, external_number, internal_number, billsec)
                          VALUES (?, ?, 1, '998935550112', '101', 0)`);
  add.run('c-2', '2026-09-20T09:00:00Z'); add.run('c-3', '2026-09-20T10:00:00Z'); add.run('c-4', '2026-09-20T11:00:00Z');
```

```js
// CRM_PHONE_V1 (2026-10-17) — ТЕЛЕФОН ВНУТРИ CRM: четыре вызова рабочего места телефониста.
//
// Владелец: «there is a easyphone, I want it to be inside the easymed, not in
// separate adress» → на вопрос, где именно: «inside the CRM».
//
// ОТКУДА ЭТО. До этого дня эти четыре вызова были маршрутами ОТДЕЛЬНОЙ программы
// EasyPhone (phone/index.js, порт 8020, EASYPHONE_V1): /api/me, /api/my-extension,
// /api/extensions, /api/calls. Программа жила на своём адресе, ссылки на неё из
// EasyMed не было, и адрес набирали руками. Теперь рабочее место — панель раздела
// «CRM · Заявки» (views/crm-phone.js), а её сервер — обычные RPC за теми же воротами.
//
// ВОРОТА ТЕ ЖЕ, ЧТО БЫЛИ, и это главное, что здесь нельзя потерять. У EasyPhone
// они стоили разбора настоящей дыры (CALLCENTER_OPERATOR_V1): журнал звонков —
// это вся история обращений клиники с номерами и записями разговоров, и право
// ПОЗВОНИТЬ его не открывает. Три права — три двери:
//   crm.dial (edit)    — набор и своя трубка;
//   crm.calls (view)   — журнал;
//   crm.recording      — ссылка на запись: её проверяет telephony_call_recording,
//                        а журнал лишь говорит флагом may_hear, рисовать ли кнопку.
// Набор и запись — telephony_dial и telephony_call_recording (rpc/telephony.js),
// как и прежде: второго разъёма не заводится.
import { grantAllows } from '../grants.js';
import { candidateExtensions, dialProvider } from '../telephony/dial.js';
import { pbxCall } from '../telephony/onlinepbx.js';
import { getProviderRow, pbxOptions, providerConfig, listProviders } from '../telephony/providers.js';
import { RpcError } from './telephony.js';

// Зеркало DIAL_ROLES / CALL_LOG_ROLES из rpc/telephony.js — ПРЕЖНЕЕ ПОВЕДЕНИЕ для
// ролей, которым эти ключи ещё не настраивали (CALLCENTER_OPERATOR_V1). Иначе двери
// разошлись бы в день, когда клиника выдала телефон своей роли: кнопка на доске
// работала бы, а панель отвечала «недоступна».
export const PHONE_ROLES = Object.freeze(['admin', 'registrar', 'callcenter']);

const mayDial = (db, user) => grantAllows(db, user, 'crm.dial', 'edit', PHONE_ROLES);
const mayLog  = (db, user) => grantAllows(db, user, 'crm.calls', 'view', PHONE_ROLES);
const mayHear = (db, user) => grantAllows(db, user, 'crm.recording', 'edit', PHONE_ROLES);

// Фразы отказов — те же, что стояли на дверях EasyPhone; в словаре i18n они уже есть.
const DENY_DIAL = 'Звонить из программы можно тем, кому выдано право «Позвонить пациенту» («Настройки → Роли»).';
const DENY_LOG = 'Журнал звонков открыт тем, кому выдано право «Журнал звонков» («Настройки → Роли»).';
const DENY_WORKPLACE = 'Программа телефонии доступна тем, кому выдано право звонить из программы или смотреть журнал звонков («Настройки → Роли»).';

/**
 * Кто я и чем звоню. Открыто тому, у кого есть ХОТЬ ОДНО из двух прав: человек,
 * которому выдали журнал без набора, иначе упирался бы в «недоступно» на пустой
 * панели — при открытом ему журнале. Три флага — чтобы экран рисовал только то,
 * что сервер потом не откажет.
 */
export function telephonyPhoneMe(db, _args, user) {
  const dial = mayDial(db, user);
  const log = mayLog(db, user);
  if (!dial && !log) throw new RpcError(DENY_WORKPLACE, 403);
  const me = db.prepare('SELECT id, full_name, pbx_extension FROM users WHERE id = ?').get(user && user.id ? user.id : 0) || {};
  const via = dialProvider(db);
  return {
    id: me.id ?? null,
    full_name: me.full_name || '',
    extension: me.pbx_extension || '',
    line: via ? via.kind : null,
    // Живые трубки из журнала — из них оператор выбирает свою, если станция молчит.
    seen_extensions: dial ? candidateExtensions(db, 12) : [],
    may_dial: dial,
    may_log: log,
    // Слушать записи, не видя журнала, всё равно неоткуда.
    may_hear: log && mayHear(db, user),
  };
}

/**
 * Мой внутренний номер. Пишет ТОЛЬКО себе: чужую трубку назначать нельзя — иначе
 * звонок уйдёт от чужого имени, и разбор смены соврёт.
 */
export function telephonyMyExtension(db, args, user) {
  if (!mayDial(db, user)) throw new RpcError(DENY_DIAL, 403);
  const ext = String((args && args.extension) || '').trim();
  if (ext && !/^[0-9*#]{1,12}$/.test(ext)) throw new RpcError('Внутренний номер — это цифры, не длиннее двенадцати.', 400);
  db.prepare('UPDATE users SET pbx_extension = ? WHERE id = ?').run(ext || null, user.id);
  return { ok: true, extension: ext };
}

/**
 * Внутренние номера у самой станции — чтобы выбирать из настоящего, а не вспоминать.
 * Только onlinePBX отдаёт такой список; у других линий — пусто, и экран берёт
 * seen_extensions. Отказ станции — словом в error, а не исключением: пустой список
 * не беда, «не выбран» работает.
 */
export async function telephonyExtensions(db, _args, user, { pbxCallImpl = pbxCall } = {}) {
  if (!mayDial(db, user)) throw new RpcError(DENY_DIAL, 403);
  const line = (listProviders(db) || []).find((p) => p.enabled && p.kind === 'onlinepbx');
  if (!line) return { extensions: [], error: '' };
  const row = getProviderRow(db, line.id);
  const r = await pbxCallImpl(providerConfig(row).domain, 'user/get.json', {}, pbxOptions(db, row));
  const list = r.ok && Array.isArray(r.data)
    ? r.data.filter((u) => u && u.enabled !== false).map((u) => String(u.num))
    : [];

/**
 * Журнал — один запрос на весь экран: последние звонки с именем пациента и именем
 * оператора (по внутреннему номеру). Панель спрашивает его каждые несколько секунд,

 */
export function telephonyCallLog(db, args, user) {
  if (!mayLog(db, user)) throw new RpcError(DENY_LOG, 403);
  const limit = Math.max(1, Math.min(200, Number(args && args.limit) || 60));
  const hear = mayHear(db, user);
  const rows = db.prepare(`
      SELECT c.id, c.started_at, c.call_type, c.external_number, c.internal_number,
             c.waitsec, c.billsec, c.disposition, c.recording_url, c.patient_id,
             p.full_name AS patient_name, p.mrn AS patient_mrn,
             u.full_name AS operator_name
        FROM calls c
        LEFT JOIN patients p ON p.id = c.patient_id
        LEFT JOIN users u ON u.pbx_extension IS NOT NULL AND u.pbx_extension <> ''
                         AND u.pbx_extension = c.internal_number
       ORDER BY c.started_at DESC, c.id DESC
       LIMIT ?`).all(limit);
  return {
    may_hear: hear,
    may_dial: mayDial(db, user),
    calls: rows.map((c) => ({ ...c, recording_url: hear ? c.recording_url : null, has_recording: !!c.recording_url })),
  };
}
```

```js
// CRM_PHONE_V1 (2026-10-17) — РЕШЕНИЯ ПАНЕЛИ «ТЕЛЕФОН» БЕЗ DOM.
//
// Что показывать на дисплее набора, в какую группу дня положить звонок, как
// фильтровать журнал, каким словом назвать исход, когда загорается полоса «только
// что звонили» и какая строка журнала — наш только что набранный звонок. Ни DOM,
// ни часов, ни сети: всё приходит параметрами, и всё проверяется в
// crm-phone-logic.test.js на граничных значениях (тот же договор, что у
// telephony-logic.js рядом).
//
// Слова исходов — ключи словаря i18n-strings.js: панель рисует их через h(),
// текстовый путь которого переводит tr(). Длительность и исход вендора берутся из
// telephony-logic.js — второго словаря «BUSY → Абонент занят» в коде быть не должно.
import { formatDuration, dispositionLabel } from './telephony-logic.js';
import { nameKey, phoneKey } from './views/crm-phone-match.js';

/** Короче семи цифр не бывает ни одного набираемого номера; тот же порог у сервера (telephony/dial.js). */
export const MIN_DIGITS = 7;
/** «Только что звонили»: входящий без ответа моложе двух минут (эвристика «live» из EasyPhone). */
export const JUST_CALLED_MS = 2 * 60 * 1000;
/** Строка журнала о нашем звонке может начаться чуть РАНЬШЕ нашего нажатия: часы станции — не наши. */
export const DIAL_MATCH_GRACE_MS = 60 * 1000;
/** Карточка «трубка звонит» без ответа журнала живёт десять минут и уступает покою. */
export const DIAL_CARD_TTL_MS = 10 * 60 * 1000;
/** Экран телефониста обновляется сам — он смотрит на него, а не жмёт F5. */
export const POLL_MS = 7000;
const TAIL = 9;

export const digitsOf = (v) => String(v == null ? '' : v).replace(/\D+/g, '');
export const tail9 = (v) => digitsOf(v).slice(-TAIL);

/** Набранное: цифры, * и #; плюс — только ведущий («99+8» станция не наберёт). */
export function cleanTyped(s) {
    const t = String(s || '').replace(/[^\d*#+]/g, '');
    return t.startsWith('+') ? '+' + t.slice(1).replace(/\+/g, '') : t.replace(/\+/g, '');
}

/**
 * Показ набранного: РОВНО ТО, ЧТО НАБРАЛИ, группами по три. Общий формат телефонов
 * (phone-format.js) дописывает «+» и код страны — в карточке это удобство, а в
 * наборнике враньё: человек увидел бы не тот номер, который уйдёт на станцию.
 */
export function groupTyped(v) {
    const s = cleanTyped(v);
    const plus = s.startsWith('+');
    const body = plus ? s.slice(1) : s;
    return (plus ? '+' : '') + body.replace(/(.{3})(?=.)/g, '$1 ');
}

export const canDial = (v) => digitsOf(v).length >= MIN_DIGITS;

const ymd = (d) => d.getFullYear() + '-' + d.getMonth() + '-' + d.getDate();
const pad2 = (n) => String(n).padStart(2, '0');

/** Группа дня по ЛОКАЛЬНОЙ дате клиники: 'today' | 'yesterday' | 'earlier'. Мусор — 'earlier'. */
export function dayGroup(iso, now = new Date()) {
    const d = new Date(iso);
    if (Number.isNaN(d.getTime())) return 'earlier';
    if (ymd(d) === ymd(now)) return 'today';
    const y = new Date(now.getTime());
    y.setDate(y.getDate() - 1);
    return ymd(d) === ymd(y) ? 'yesterday' : 'earlier';
}

export const GROUP_ORDER = ['today', 'yesterday', 'earlier'];

/** Звонки по группам дня в порядке «сегодня · вчера · ранее»; пустые группы опущены. */
export function groupCalls(calls, now = new Date()) {
    const buckets = { today: [], yesterday: [], earlier: [] };
    for (const c of calls || []) buckets[dayGroup(c && c.started_at, now)].push(c);
    return GROUP_ORDER.filter((k) => buckets[k].length).map((k) => ({ key: k, rows: buckets[k] }));
}

/** Время строки: сегодня — ЧЧ:ММ, иначе ДД.ММ ЧЧ:ММ. */
export function timeLabel(iso, now = new Date()) {
    const d = new Date(iso);
    if (Number.isNaN(d.getTime())) return '—';
    const t = pad2(d.getHours()) + ':' + pad2(d.getMinutes());
    return ymd(d) === ymd(now) ? t : pad2(d.getDate()) + '.' + pad2(d.getMonth() + 1) + ' ' + t;
}

/**
 * Исход звонка словами клиники. Разговор был — «Отвечен» в любую сторону; входящий
 * без разговора — «Пропущен»; наш недозвон — словом станции (BUSY → «Абонент
 * занят», CANCEL → «Отменён»), а без слова — «Без ответа».
 */
export function outcomeOf(c) {
    const out = Number(c && c.call_type) === 1;
    const answered = Number(c && c.billsec) > 0;
    const kind = out ? 'out' : (answered ? 'in' : 'miss');
    let word;
    if (answered) word = 'Отвечен';
    else if (!out) word = 'Пропущен';
    else {
        const d = String((c && c.disposition) || '').trim().toUpperCase();
        word = d && d !== 'ANSWER' ? dispositionLabel(d) : 'Без ответа';
    }
    return { out, answered, kind, word, icon: kind === 'miss' ? 'PhoneMissed' : (out ? 'PhoneOut' : 'PhoneIn') };
}

/** Длительность разговора m:ss; не было разговора — тире. */
export const talkLabel = (sec) => (Number(sec) > 0 ? formatDuration(Number(sec)) : '—');

/** Запрос — это номер, если в нём нет ничего, кроме цифр, пробелов, плюса, скобок и дефисов. */
const looksLikeNumber = (q) => /^[\d\s+()\-]+$/.test(q);

export function matchesQuery(c, query) {
    const q = String(query || '').trim();
    if (!q) return true;
    if (looksLikeNumber(q)) return digitsOf(c && c.external_number).includes(digitsOf(q));
    return nameKey(c && c.patient_name).includes(nameKey(q));
}

/**
 * Сегменты как в образце: «Входящие» — ВСЕ входящие (и пропущенные тоже),
 * «Исходящие» — все наши, «Пропущенные» — входящие без разговора. Недозвон
 * исходящего пропущенным не считается: пациент нам не звонил.
 */
export function filterCalls(calls, { seg = 'all', query = '' } = {}) {
    return (calls || []).filter((c) => {
        const o = outcomeOf(c);
        if (seg === 'in' && o.out) return false;
        if (seg === 'out' && !o.out) return false;
        if (seg === 'miss' && o.kind !== 'miss') return false;
        return matchesQuery(c, query);
    });
}

/** Полоса «только что звонили»: свежайший входящий без ответа моложе двух минут. */
export function justCalled(calls, now = Date.now()) {
    return (calls || []).find((c) => Number(c.call_type) === 0 && !(Number(c.billsec) > 0)
        && now - Date.parse(c.started_at) < JUST_CALLED_MS) || null;
}

/**
 * Какая строка журнала — наш только что набранный звонок: исходящий, хвост номера
 * из девяти цифр совпал (у одного номера код страны то есть, то нет), начался не
 * раньше чем за минуту до нашего нажатия.
 */
export function matchDialed(calls, { digits, since }) {
    const t = tail9(digits);
    if (!t) return null;
    return (calls || []).find((c) => Number(c.call_type) === 1 && tail9(c.external_number) === t
        && Date.parse(c.started_at) >= Number(since) - DIAL_MATCH_GRACE_MS) || null;
}

/** Инициалы для аватара карточки звонка; без имени — «#». */
export function initialsOf(name) {
    const parts = String(name || '').trim().split(/\s+/).filter(Boolean);
    return parts.slice(0, 2).map((w) => w[0]).join('').toUpperCase() || '#';
}

/**
 * Кому принадлежит номер: заявка с доски (по ключу номера, как ищет сама доска),
 * иначе пациент из строки журнала с тем же номером. По обрывку не угадываем.
 */
export function whoIs(digits, { leads = [], calls = [] } = {}) {
    const d = digitsOf(digits);
    if (d.length < MIN_DIGITS) return null;
    const k = phoneKey(d);
    const lead = (leads || []).find((r) => r && phoneKey(r.phone) === k);
    if (lead) return { kind: 'lead', name: lead.full_name || '', lead };
    const c = (calls || []).find((x) => x && x.patient_name && phoneKey(x.external_number) === k);
    if (c) return { kind: 'patient', name: c.patient_name, patient_id: c.patient_id || null };
    return null;
}
```
